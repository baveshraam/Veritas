# Build constraints for the AppSail image (see Dockerfile).
#
# torch is pinned to the CPU build and MUST stay pinned during the main resolve, not
# merely pre-installed: `data[translation]` asks for `torch>=2.2`, and left to itself
# pip happily satisfies that from PyPI with the CUDA build — dragging nvidia-cudnn /
# triton (~2GB) into a container that has no GPU, and thrashing the resolver while it
# backtracks between the two indexes.
torch==2.6.0+cpu

# psycopg 3.2.0's `binary` extra requires psycopg-binary==3.2.0.dev1 — a version that
# was never published. Any resolve that backtracks onto 3.2.0 dies there. Pin past it.
psycopg==3.2.9

# langgraph builds its state schemas with TypedDict(extra_items=...), which needs
# typing_extensions 4.13+ — but its own floor is looser, and torch's resolve used to
# drag the tree down to 4.12.2, which crashed every /chat query at graph-build time.
typing_extensions>=4.13
